(function(){
  var EPOCH_MS = Date.UTC(2026,0,1);      // эпо́ха: 1 января 2026, 00:00 UTC
  var DYEAR = 364, DMONTH = 28;            // D-исчисление: 13 месяцев по 28
  var FIRST = 18.5 * DYEAR;                // первое пришествие: 6734 D-суток от эпо́хи
  var INTERVAL = 44.8 * DYEAR;             // цикл Всадника: 16307.2 D-суток
  var MS_DAY = 86400000;

  function pad(n){ return (n<10?'0':'')+n; }

  function tick(){
    var now = Date.now();
    var d = (now - EPOCH_MS) / MS_DAY;     // ГЛАВНОЕ ИСПРАВЛЕНИЕ: мс -> сутки
    if (d < 0) d = 0;                       // защита от перевранных часов

    // подбираем виток цикла пришествий
    var k = 0, next = FIRST;
    while (next <= d){ k++; next = FIRST + k*INTERVAL; }
    var rem = next - d;
    var ry = Math.floor(rem / DYEAR);
    var rm = Math.floor((rem - ry*DYEAR) / DMONTH);
    var rd = Math.floor(rem - ry*DYEAR - rm*DMONTH);
    document.getElementById('cdLeft').textContent =
      'ДО КАТАСТРОФЫ: ' + ry + ' D-лет ' + rm + ' D-мес ' + rd + ' D-дн';

    // земная дата пришествия — ручное форматирование, никаких NaN
    var cata = new Date(EPOCH_MS + next*MS_DAY);
    document.getElementById('cdDate').textContent =
      pad(cata.getUTCDate()) + '.' + pad(cata.getUTCMonth()+1) + '.' + cata.getUTCFullYear();

    // следующие витки цикла
    var arr = [];
    for (var i = 1; i <= 3; i++){
      arr.push(new Date(EPOCH_MS + (FIRST + (k+i)*INTERVAL)*MS_DAY).getUTCFullYear());
    }
    document.getElementById('cdNext').textContent = arr.join(' · ');
  }
  tick();
  setInterval(tick, 60000);   // пересчёт раз в минуту
})();
</script>
